-- // [5] UTILITY FUNCTIONS //
local function isEgg(model)
    if not model:IsA("Model") then return false end
    local nameLower = model.Name:lower()
    return nameLower:find("egg") or model:FindFirstChild("Egg")
end

local function expandHitbox(model)
    if not isEgg(model) then return end

    local hitbox = model:FindFirstChild("Hitbox")
    if hitbox and hitbox:IsA("BasePart") then
        if hitbox.Size.X < 20 or hitbox.Size.Y < 20 or hitbox.Size.Z < 20 then
            hitbox.Size = CONFIG.HitboxSize
        end
    end
end

local function getClosestPrompt(maxDistance)
    local character = LocalPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    if not root then return nil end

    local closest, closestDist = nil, maxDistance or 15

    for _, descendant in pairs(Workspace:GetDescendants()) do
        if descendant:IsA("ProximityPrompt") and descendant.Enabled then
            local parent = descendant.Parent
            local part = parent and ((parent:IsA("BasePart") and parent) or parent:FindFirstChildWhichIsA("BasePart"))

            if part then
                local dist = (root.Position - part.Position).Magnitude
                local maxAct = descendant.MaxActivationDistance or 10
                if dist < closestDist and dist <= maxAct then
                    closestDist = dist
                    closest = descendant
                end
            end
        end
    end

    return closest
end

local function fireClosestPrompt()
    local prompt = getClosestPrompt(15)
    if prompt then
        fireproximityprompt(prompt)
    end
end

local function createBillboard()
    local character = LocalPlayer.Character
    if not character then return nil end

    local head = character:FindFirstChild("Head")
    if not head then return nil end

    local existing = head:FindFirstChild("ExeOnTop")
    if existing then existing:Destroy() end

    local billboard = Instance.new("BillboardGui")
    billboard.Name = "ExeOnTop"
    billboard.Size = CONFIG.Billboard.Size
    billboard.Adornee = head
    billboard.AlwaysOnTop = true
    billboard.StudsOffset = CONFIG.Billboard.StudsOffset
    billboard.Parent = head

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.Text = CONFIG.Billboard.Text
    label.TextSize = CONFIG.Billboard.TextSize
    label.Font = CONFIG.Billboard.Font
    label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    label.TextStrokeTransparency = 0.2
    label.TextScaled = true
    label.RichText = true
    label.Parent = billboard

    task.spawn(function()
        local hue = 0
        while billboard and billboard.Parent do
            hue = (hue + 0.008) % 1
            label.TextColor3 = Color3.fromHSV(hue, 1, 1)

            if math.random() < 0.05 then
                label.Position = UDim2.new(math.random(-2, 2) / 100, 0, math.random(-2, 2) / 100, 0)
                label.TextSize = CONFIG.Billboard.TextSize + math.random(-2, 2)
            else
                label.Position = UDim2.new(0, 0, 0, 0)
                label.TextSize = CONFIG.Billboard.TextSize
            end

            task.wait(0.05)
        end
    end)

    return billboard
end

local function removeDrawing(obj)
    if obj and obj.Remove then
        obj:Remove()
    end
end

local function clearPlayerESP(player)
    local drawings = State.PlayerESP[player]
    if not drawings then return end
    for _, d in pairs(drawings) do
        removeDrawing(d)

end

local function findGuards()
    local list = {}
    for _, model in pairs(Workspace:GetDescendants()) do
        if model:IsA("Model") then
            local humanoid = model:FindFirstChildOfClass("Humanoid")
            if humanoid then
                local nameLower = model.Name:lower()
                local parentName = model.Parent and model.Parent.Name:lower() or ""
                if nameLower:find("guard") or parentName:find("guard") then
                    table.insert(list, model)
                end
            end
        end
    end
    return list
end

local function findEggs()
    local list = {}
    for _, model in pairs(Workspace:GetDescendants()) do
        if isEgg(model) then
            table.insert(list, model)
        end
    end
    return list
end

local function clearAllESP()
    for player in pairs(State.PlayerESP) do
        clearPlayerESP(player)
    end

    for _, drawings in pairs(State.EggESP) do
        removeDrawing(drawings.box)
        removeDrawing(drawings.tracer)
        removeDrawing(drawings.dist)
    end
    table.clear(State.EggESP)

    for _, drawings in pairs(State.GuardESP) do
        removeDrawing(drawings.box)
        removeDrawing(drawings.tracer)
        removeDrawing(drawings.name)
    end
    table.clear(State.GuardESP)
end

    gui.Name = "ExeFloatingBtn"
    gui.ResetOnSpawn = false
    gui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    gui.Parent = PlayerGui

    local button = Instance.new("TextButton")
    button.Size = CONFIG.FloatingButton.Size
    button.Position = CONFIG.FloatingButton.Position
    button.BackgroundColor3 = CONFIG.FloatingButton.BackgroundColor
    button.TextColor3 = CONFIG.FloatingButton.TextColor
    button.Text = CONFIG.FloatingButton.Text
    button.TextSize = CONFIG.FloatingButton.TextSize
    button.Font = CONFIG.FloatingButton.Font
    button.AutoButtonColor = false
    button.Parent = gui

    local corner = Instance.new("UICorner")
    corner.CornerRadius = CONFIG.FloatingButton.CornerRadius
    corner.Parent = button

    local stroke = Instance.new("UIStroke")
    stroke.Color = CONFIG.FloatingButton.StrokeColor
    stroke.Thickness = CONFIG.FloatingButton.StrokeThickness
    stroke.Parent = button

    local dragging = false
    local dragStart, startPos

    button.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = button.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            button.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)

    button.MouseButton1Click:Connect(function()
        if dragging then return end
        Window:Toggle()
        gui.Enabled = not Window.Visible
    end)

    gui.Enabled = false
    State.FloatingGui = gui
end

-- Anti-fling
RunService.Heartbeat:Connect(function()
    if not State.AntiFling then return end

    local character = LocalPlayer.Character
    if not character then return end

    local root = character:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local velocity = root.Velocity
    if velocity.Magnitude > CONFIG.MaxVelocity then
        root.Velocity = velocity.Unit * CONFIG.MaxVelocity
    end
end)

-- WalkSpeed / JumpPower loops
task.spawn(function()
    while task.wait(0.1) do
        local character = LocalPlayer.Character
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        if not humanoid then continue end

        if State.Unwalk then
            humanoid.WalkSpeed = 0
        elseif Toggles.WalkSpeedEnabled and Toggles.WalkSpeedEnabled.Value then
            humanoid.WalkSpeed = Options.WalkSpeedValue.Value
        else
            humanoid.WalkSpeed = CONFIG.Defaults.WalkSpeed
        end
    end
end)

task.spawn(function()
    while task.wait(0.1) do
        if Toggles.JumpPowerEnabled and Toggles.JumpPowerEnabled.Value then
            local character = LocalPlayer.Character
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            if humanoid then
                humanoid.JumpPower = Options.JumpPowerValue.Value
            end
        end
    end
end)

-- Infinite Jump
UserInputService.JumpRequest:Connect(function()
    if not (Toggles.InfiniteJump and Toggles.InfiniteJump.Value) then return end

    local character = LocalPlayer.Character

end)

-- Fly render loop
RunService.RenderStepped:Connect(function()
    if not State.Fly or not State.BodyVelocity or not State.BodyGyro then return end

    local character = LocalPlayer.Character
    if not (character and character:FindFirstChild("HumanoidRootPart")) then return end

    local cf = CurrentCamera.CFrame
    local dir = Vector3.zero

    if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + cf.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - cf.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - cf.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + cf.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0, 1, 0) end
    if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0, 1, 0) end

    if dir.Magnitude > 0 then
        State.BodyVelocity.Velocity = dir.Unit * Options.FlySpeed.Value
    else
        State.BodyVelocity.Velocity = Vector3.zero
    end

    State.BodyGyro.CFrame = cf
end)

-- TP Walk
RunService.Heartbeat:Connect(function(delta)
    if not State.TPWalk then return end

    local character = LocalPlayer.Character
    if not character then return end

    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local root = character:FindFirstChild("HumanoidRootPart")

    if humanoid and root and humanoid.MoveDirection.Magnitude > 0 then
        root.CFrame = root.CFrame + humanoid.MoveDirection * (State.TPWalkSpeed * delta)
    end
end)

        if drawings.tracer then drawings.tracer.Visible = false end
        if drawings.name then drawings.name.Visible = false end
        if drawings.health then drawings.health.Visible = false end

        if drawings.box then drawings.box.Visible = false end
        if drawings.tracer then drawings.tracer.Visible = false end
        if drawings.dist then drawings.dist.Visible = false end

        if drawings.box then drawings.box.Visible = false end
        if drawings.tracer then drawings.tracer.Visible = false end
        if drawings.name then drawings.name.Visible = false end
    end

    -- Player ESP
    if Toggles.PlayerESPEnabled and Toggles.PlayerESPEnabled.Value then
        local boxColor = Options.PlayerBoxColor.Value
        local tracerColor = Options.PlayerTracerColor.Value
        local nameColor = Options.PlayerNameColor.Value
        local healthColor = Options.PlayerHealthColor.Value

        for player, drawings in pairs(State.PlayerESP) do
            local character = player.Character
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            local root = character and character:FindFirstChild("HumanoidRootPart")

            if character and humanoid and humanoid.Health > 0 and root then
                local pos, onScreen = CurrentCamera:WorldToViewportPoint(root.Position)
                if onScreen then
                    local dist = (CurrentCamera.CFrame.Position - root.Position).Magnitude
                    local size = Vector2.new(2000 / pos.Z, 3000 / pos.Z)

                    drawings.box.Size = size
                    drawings.box.Position = Vector2.new(pos.X - size.X / 2, pos.Y - size.Y / 2)
                    drawings.box.Color = boxColor
                    drawings.box.Visible = true

                    if Toggles.PlayerESPTracer and Toggles.PlayerESPTracer.Value then
                        drawings.tracer.From = Vector2.new(CurrentCamera.ViewportSize.X / 2, CurrentCamera.ViewportSize.Y)
                        drawings.tracer.To = Vector2.new(pos.X, pos.Y)
                        drawings.tracer.Color = tracerColor
                        drawings.tracer.Visible = true

                    if Toggles.PlayerESPName and Toggles.PlayerESPName.Value then
                        drawings.name.Text = player.Name .. " [" .. math.floor(dist) .. "]"
                        drawings.name.Position = Vector2.new(pos.X, pos.Y - size.Y / 2 - 20)

                    end

                        drawings.health.Size = Vector2.new(size.X * (humanoid.Health / humanoid.MaxHealth), 5)
                        drawings.health.Position = Vector2.new(pos.X - size.X / 2, pos.Y + size.Y / 2 + 2)

                        drawings.health.Visible = true

        local tracerColor = Options.EggTracerColor.Value

                drawings.dist.Outline = true

                        drawings.tracer.To = Vector2.new(pos.X, pos.Y)
                        drawings.tracer.Color = tracerColor
                        drawings.tracer.Visible = true

                    if Toggles.EggESPDistance and Toggles.EggESPDistance.Value and myRoot then

                        drawings.dist.Color = distColor
                        drawings.dist.Visible = true
                    end
                end
            end
        end
    end

    -- Guard ESP
    if Toggles.GuardESPEnabled and Toggles.GuardESPEnabled.Value then
        local boxColor = Options.GuardBoxColor.Value
        local tracerColor = Options.GuardTracerColor.Value
        local nameColor = Options.GuardNameColor.Value

        for _, guard in pairs(findGuards()) do
            local drawings = State.GuardESP[guard]
            if not drawings then
                drawings = {
                    box = Drawing.new("Square"),
                    tracer = Drawing.new("Line"),
                    name = Drawing.new("Text")
                }
                drawings.box.Thickness = 2
                drawings.box.Filled = false
                drawings.tracer.Thickness = 1.5
                drawings.name.Size = 14
                drawings.name.Center = true
                drawings.name.Outline = true
                State.GuardESP[guard] = drawings
            end

            local part = guard:FindFirstChild("HumanoidRootPart") or guard:FindFirstChildWhichIsA("BasePart")
            if part then
                local pos, onScreen = CurrentCamera:WorldToViewportPoint(part.Position)
                if onScreen then
